let allCurrentVerses=[];

function copyVerse(idx,btn){
  const a=allCurrentVerses[idx];
  if(!a)return;
  const txt=a.text+' ﴿'+a.numberInSurah+'﴾';
  if(navigator.clipboard&&navigator.clipboard.writeText){
    navigator.clipboard.writeText(txt).then(()=>{done(btn);}).catch(()=>{fallback(txt,btn);});
  }else{fallback(txt,btn);}
  function done(b){b.textContent='✅ تم النسخ';b.classList.add('copied');setTimeout(()=>{b.textContent='📋 نسخ';b.classList.remove('copied');},2000);}
  function fallback(t,b){
    const ta=document.createElement('textarea');ta.value=t;ta.style.position='fixed';ta.style.opacity='0';
    document.body.appendChild(ta);ta.focus();ta.select();
    try{document.execCommand('copy');}catch(e){}
    document.body.removeChild(ta);done(b);
  }
}

  document.getElementById('versesContainer').innerHTML=list.map((a,i)=>`

        <span class="verse-num">﴿${a.numberInSurah}﴾</span>
        <button class="copy-verse-btn" onclick="copyVerse(${allCurrentVerses.indexOf(a)},this)">📋 نسخ</button>
      </div>
    </div>`).join('');
}

  'ar.alafasy':    {server:'https://server8.mp3quran.net/afs',pad:3},
  'ar.sudais':     {server:'https://server11.mp3quran.net/sds',pad:3},
  'ar.husary':     {server:'https://server13.mp3quran.net/husr',pad:3},
  'ar.abdulbasit': {server:'https://server7.mp3quran.net/basit',pad:3},
  'ar.maher':      {server:'https://server12.mp3quran.net/maher',pad:3},
  'ar.minshawi':   {server:'https://server10.mp3quran.net/minsh',pad:3},
  'ar.dosary':     {server:'https://download.quranicaudio.com/quran/yasser_ad-dussary',pad:3},
  'ar.ghamdi':     {server:'https://server7.mp3quran.net/s_gmd',pad:3}
};

function loadSurahAudio(){
  if(!currentSurahId)return;
  const sel=document.getElementById('reciterSelect').value;
  const rec=RECITER_MAP[sel]||RECITER_MAP['ar.alafasy'];
  const sNum=String(currentSurahId).padStart(rec.pad,'0');
  const url=`${rec.server}/${sNum}.mp3`;
  const audio=document.getElementById('surahAudio');
  audio.src=url;
  audio.load();
  document.getElementById('audioStatus').textContent='اضغط ▶ للاستماع';
  audio.oncanplay=()=>{
    document.getElementById('audioStatus').textContent='';
  };
  audio.onerror=()=>{
    document.getElementById('audioStatus').textContent='⚠️ تعذّر تحميل الصوت — جرّب قارئاً آخر';
  };
  audio.onended=()=>{
    document.getElementById('audioStatus').textContent='✅ انتهت السورة';
  };
}

function changeReciter(){if(currentSurahId)loadSurahAudio();}

async function openSurah(id){
  const surah=SURAHS.find(s=>s.id===id);
  if(!surah)return;
  currentSurahId=id;
  allCurrentVerses=[];
  document.getElementById('verseSearch').value='';
  document.getElementById('surahModalName').textContent=surah.name;
  document.getElementById('surahModalSub').textContent=surah.nameEn+' • '+surah.verses+' '+t('verse');
  document.getElementById('versesContainer').innerHTML='<div class="loading">'+t('loading')+'</div>';
  document.getElementById('surahModal').classList.add('open');
  // load audio
  loadSurahAudio();
  // load text
  try{
    const res=await fetch('https://api.alquran.cloud/v1/surah/'+id);
    const data=await res.json();
    if(data.code===200){
      allCurrentVerses=data.data.ayahs;
      renderVerses(allCurrentVerses);
    }else throw new Error();
  }catch{
    const fallback={1:[
      {text:"الْحَمْدُ لِلَّهِ رَبِّ الْعَالَمِينَ",numberInSurah:1},
      {text:"الرَّحْمَٰنِ الرَّحِيمِ",numberInSurah:2},
      {text:"مَالِكِ يَوْمِ الدِّينِ",numberInSurah:3},
      {text:"إِيَّاكَ نَعْبُدُ وَإِيَّاكَ نَسْتَعِينُ",numberInSurah:4},
      {text:"اهْدِنَا الصِّرَاطَ الْمُسْتَقِيمَ",numberInSurah:5},
      {text:"صِرَاطَ الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ غَيْرِ الْمَغْضُوبِ عَلَيْهِمْ وَلَا الضَّالِّينَ",numberInSurah:6}
    ]};
    allCurrentVerses=fallback[id]||[];
    if(allCurrentVerses.length>0){renderVerses(allCurrentVerses);}
    else{document.getElementById('versesContainer').innerHTML='<div class="loading" style="font-size:13px">تعذّر التحميل — تحقق من اتصالك</div>';}
  }
}

// ══════ PRAYER COUNTDOWN ══════
function getNextPrayer(){
  const c=CITIES[selectedCity];
  const prayers=[
    {name:t('fajr'),time:c.fajr},
    {name:t('dhuhr'),time:c.dhuhr},
    {name:t('asr'),time:c.asr},
    {name:t('maghrib'),time:c.maghrib},
    {name:t('isha'),time:c.isha}
  ];
  const now=new Date();
  const nowMins=now.getHours()*60+now.getMinutes();
  for(const p of prayers){
    const[h,m]=p.time.split(':').map(Number);
    const pMins=h*60+m;
    if(pMins>nowMins) return {name:p.name,h,m,pMins};
  }
  // wrap to fajr next day
  const[h,m]=c.fajr.split(':').map(Number);
  return {name:t('fajr'),h,m,pMins:h*60+m+1440};
}

function updateCountdown(){
  const next=getNextPrayer();
  const now=new Date();
  const nowSecs=now.getHours()*3600+now.getMinutes()*60+now.getSeconds();
  let targetSecs=next.h*3600+next.m*60;
  if(targetSecs<=now.getHours()*3600+now.getMinutes()*60) targetSecs+=86400;
  let diff=targetSecs-nowSecs;
  if(diff<0) diff+=86400;
  const hh=Math.floor(diff/3600);
  const mm=Math.floor((diff%3600)/60);
  const ss=diff%60;
  const pad=n=>String(n).padStart(2,'0');
  const el=document.getElementById('countdownTimer');
  const nameEl=document.getElementById('nextPrayerName');
  const timeEl=document.getElementById('nextPrayerTime');
  if(el) el.textContent=`${pad(hh)}:${pad(mm)}:${pad(ss)}`;
  if(nameEl) nameEl.textContent=next.name;
  const c=CITIES[selectedCity];
  const prayers=[{name:t('fajr'),time:c.fajr},{name:t('dhuhr'),time:c.dhuhr},{name:t('asr'),time:c.asr},{name:t('maghrib'),time:c.maghrib},{name:t('isha'),time:c.isha}];
  const p=prayers.find(x=>x.name===next.name);
  if(timeEl&&p) timeEl.textContent=(currentLang==='en'?'at ':currentLang==='fr'?'à ':' الساعة ')+p.time;
}

function toggleRadio(id){
  const radio=RADIOS.find(r=>r.id===id);
  if(!radio)return;
  if(playingRadioId===id){
    audio.pause();audio.src='';playingRadioId=null;
    document.getElementById('nowPlayingBar').style.display='none';
  }else{
    playingRadioId=id;
    audio.src=radio.url;
    audio.crossOrigin='anonymous';
    audio.load();
    const p=audio.play();
    if(p!==undefined){
      p.catch(err=>{
        const bar=document.getElementById('nowPlayingBar');
        if(bar){
          bar.style.display='block';
          const lbl=bar.querySelector('.now-playing-label');
          if(lbl) lbl.textContent='⚠️ '+(currentLang==='en'?'Tap play again or check connection':currentLang==='fr'?'Appuyez à nouveau ou vérifiez la connexion':'اضغط مرة أخرى أو تحقق من الاتصال');
        }
      });
    }
    document.getElementById('nowPlayingBar').style.display='block';
    document.getElementById('nowPlayingBar').querySelector('.now-playing-label').textContent='🎵 '+(currentLang==='en'?'Live Streaming':currentLang==='fr'?'Diffusion en direct':'جاري البث المباشر');
    document.getElementById('nowPlayingName').textContent=radio.flag+' '+radio.name+' — '+radio.reader;
  }

}

function renderAthkar(){
  const list=ATHKAR[athkarTab];
  document.getElementById('athkarList').innerHTML=list.map(z=>{
    const cur=athkarCounts[z.id]||0;
    const done=athkarDone[z.id];
    return`<div class="athkar-card ${done?'done':''}" id="athkar-${z.id}">
      <div class="athkar-text">${z.text}</div>
      <div class="athkar-benefit">${z.benefit}</div>

          ${done?t('completed'):cur+' / '+z.count}
        </div>
        ${!done?`<button class="count-btn" onclick="countAthkar('${z.id}',${z.count})">+</button>`:''}
      </div>
      ${z.count>1&&!done?`<div class="progress-wrap"><div class="progress-bar" id="prog-${z.id}" style="width:${Math.min((cur/z.count)*100,100)}%"></div></div>`:''}
    </div>`;
  }).join('');
}
function countAthkar(id,max){
  if(athkarDone[id])return;
  athkarCounts[id]=(athkarCounts[id]||0)+1;
  if(athkarCounts[id]>=max){athkarDone[id]=true;renderAthkar();}
  else{
    const lbl=document.getElementById('count-label-'+id);
    if(lbl)lbl.textContent=athkarCounts[id]+' / '+max;
    const bar=document.getElementById('prog-'+id);
    if(bar)bar.style.width=Math.min((athkarCounts[id]/max)*100,100)+'%';
  }
}
function switchAthkar(tab,btn){

// ══════ RENAME MODAL ══════
const DAY_MS=86400000;
function daysLeft(ts){return Math.ceil((ts+DAY_MS*7-Date.now())/DAY_MS);}
function daysLeftUsername(ts){return Math.ceil((ts+DAY_MS*30-Date.now())/DAY_MS);}

function openRenameModal(){
  if(!user)return;
  document.getElementById('renameInput').value=user.name||'';
  document.getElementById('usernameInput').value=user.username||'';
  // Name info
  const nameChanges=user.nameChanges||0;
  const lastNameChange=user.lastNameChange||0;
  const nameEl=document.getElementById('renameNameInfo');
  if(nameChanges>=2){
    const dl=daysLeft(lastNameChange);
    if(dl>0){
      nameEl.textContent=`⏳ وصلت للحد الأقصى — يمكنك التغيير بعد ${dl} يوم`;
      nameEl.style.color='#e55';
      document.getElementById('renameInput').disabled=true;
    } else {
      // reset after 7 days
      user.nameChanges=0;
      document.getElementById('renameInput').disabled=false;
      nameEl.textContent='✅ يمكنك تغيير الاسم مرة أخرى';
      nameEl.style.color='var(--green)';
    }
  } else {
    const remaining=2-nameChanges;
    nameEl.textContent=`متبقي ${remaining} تغيير على الاسم`;
    nameEl.style.color='var(--text-muted)';
    document.getElementById('renameInput').disabled=false;
  }
  // Username info
  const lastUsernameChange=user.lastUsernameChange||0;
  const usernameEl=document.getElementById('renameUsernameInfo');
  if(lastUsernameChange&&Date.now()-lastUsernameChange<DAY_MS*30){
    const dl=daysLeftUsername(lastUsernameChange);
    usernameEl.textContent=`⏳ يمكنك تغيير اسم المستخدم بعد ${dl} يوم`;
    usernameEl.style.color='#e55';
    document.getElementById('usernameInput').disabled=true;
  } else {
    usernameEl.textContent=user.username?'يمكن تغييره مرة كل 30 يوم':'اختر اسم مستخدم فريد';
    usernameEl.style.color='var(--text-muted)';
    document.getElementById('usernameInput').disabled=false;
  }
  document.getElementById('renameModal').classList.add('open');
}
function closeRenameModal(){
  document.getElementById('renameModal').classList.remove('open');
  document.getElementById('renameInput').disabled=false;
  document.getElementById('usernameInput').disabled=false;
}
document.getElementById('renameModal').addEventListener('click',function(e){if(e.target===this)closeRenameModal();});

function saveRename(){
  if(!user)return;
  const newName=document.getElementById('renameInput').value.trim();
  const newUsername=document.getElementById('usernameInput').value.trim().replace(/^@/,'');
  let changed=false;

  // Handle name change
  if(newName && newName!==user.name && !document.getElementById('renameInput').disabled){
    const nameChanges=user.nameChanges||0;
    if(nameChanges>=2){
      const dl=daysLeft(user.lastNameChange||0);
      if(dl>0){alert(`لا يمكن تغيير الاسم — انتظر ${dl} يوم`);return;}
      user.nameChanges=0;
    }
    user.name=newName;
    user.nameChanges=(user.nameChanges||0)+1;
    user.lastNameChange=Date.now();
    changed=true;
  }

  // Handle username change
  if(!document.getElementById('usernameInput').disabled){
    const uname=newUsername?('@'+newUsername):'';
    if(uname!==user.username){
      if(newUsername&&!/^[a-zA-Z0-9_]{3,20}$/.test(newUsername)){
        alert('اسم المستخدم يجب أن يكون 3-20 حرف، أرقام أو _');return;
      }
      user.username=uname;
      user.lastUsernameChange=uname?Date.now():0;
      changed=true;
    }
  }

  if(changed){saveUserToStorage();updateUser();}
  closeRenameModal();
}
